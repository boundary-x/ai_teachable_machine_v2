Teachable Machine · 프로젝트 예제

AI 비트런 · 이미지 분류 방향 제어

학습한 이미지를 분류해 전진·후진·좌회전·우회전·정지 동작을 실행합니다.

  1. Teachable Machine 클래스 이름을 forward / backward / left / right / stop으로 설정하세요.
  2. MakeCode에 bluetooth와 servos 확장을 추가하세요.
  3. 아래 코드를 복사해 MakeCode의 JavaScript 화면에 전체 붙여넣기 하세요.
  4. 블루투스 페어링을 ‘필요하지 않음’으로 설정하고 다운로드하세요.

기본 속도는 50입니다. stop 또는 다른 문자열을 받거나 블루투스 연결이 해제되면 정지합니다. 다음 명령을 받을 때까지 현재 동작을 유지합니다.

테스트용 예제 · 실물 동작과 블록 변환 확인 전

// Teachable Machine 클래스: forward, backward, left, right, stop
// 확장: bluetooth, servos / 왼쪽 바퀴 P1, 오른쪽 바퀴 P2
function 비트런_전진 () {
    servos.P1.run(50)
    servos.P2.run(-50)
}
function 비트런_후진 () {
    servos.P1.run(-50)
    servos.P2.run(50)
}
function 비트런_좌회전 () {
    servos.P1.run(-50)
    servos.P2.run(-50)
}
function 비트런_우회전 () {
    servos.P1.run(50)
    servos.P2.run(50)
}
function 비트런_정지 () {
    servos.P1.stop()
    servos.P2.stop()
}
bluetooth.onUartDataReceived(serial.delimiters(Delimiters.NewLine), function () {
    받은문자 = bluetooth.uartReadUntil(serial.delimiters(Delimiters.NewLine))
    if (받은문자 == "forward") {
        비트런_전진()
    } else if (받은문자 == "backward") {
        비트런_후진()
    } else if (받은문자 == "left") {
        비트런_좌회전()
    } else if (받은문자 == "right") {
        비트런_우회전()
    } else {
        비트런_정지()
    }
})
bluetooth.onBluetoothConnected(function () {
    비트런_정지()
})
bluetooth.onBluetoothDisconnected(function () {
    비트런_정지()
})
let 받은문자 = ""
servos.P1.setPulse(1500)
servos.P2.setPulse(1500)
비트런_정지()
bluetooth.startUartService()
